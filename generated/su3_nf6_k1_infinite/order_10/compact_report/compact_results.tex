\maketitle
At infinite coupling the enhanced symmetry is $\mathrm{SU}(7)\times\mathrm{U}(1)_q$.
The expansion is truncated at $O(t^{11})$.  We define
$[a_1,\ldots,a_{6}]
:=[a_1,\ldots,a_{6}]_{A6}$.
Dimension evaluation is the ring homomorphism
$\dim_{\mathrm{SU}(7)}:R(\mathrm{SU}(7))\to\mathbb{Z}$, with
$H_{\mathrm{dim}}(t,q)=\dim_{\mathrm{SU}(7)}H(t,q;\mathrm{SU}(7))$
and $H(t)=H_{\mathrm{dim}}(t,1)$.  Likewise,
$\operatorname{PL}_{\mathrm{dim}}(t,q)=\dim_{\mathrm{SU}(7)}
\operatorname{PL}[H(t,q;\mathrm{SU}(7))]$ and
$\operatorname{PL}[H(t)]=\operatorname{PL}_{\mathrm{dim}}(t,1)$;
the latter agrees with the independently stored scalar result.

\section{Highest-weight generating function}
\begin{equation}
\mathrm{HWG}=\PE\!\left[(\mu_1\mu_6+1)t^2+(q\mu_3+q^{-1}\mu_4)t^3+\mu_2\mu_5t^4\right].
\end{equation}
\begin{equation}
\resizebox{0.98\linewidth}{!}{${\displaystyle
\mathrm{HWG}=\frac{1}{(1-t^2)(1-\mu_1\mu_6t^2)(1-q\mu_3t^3)(1-q^{-1}\mu_4t^3)(1-\mu_2\mu_5t^4)}.}$}
\end{equation}
