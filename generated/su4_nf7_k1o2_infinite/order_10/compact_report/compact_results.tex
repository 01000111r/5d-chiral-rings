The flavour symmetry is $\mathrm{SU}(7)\times\mathrm{SU}(2)\times\mathrm{U}(1)_{q}$.
The expansion is truncated at $O(t^{11})$.  We define
$[a_1,\ldots,a_{6};b]:=[a_1,\ldots,a_{6}]_{A6}\otimes[b]_{A1^{(1)}}$.
Dimension evaluation is the ring homomorphism
$\dim_{\mathrm{SU}(7)\times\mathrm{SU}(2)}:R(\mathrm{SU}(7)\times\mathrm{SU}(2))\to\mathbb{Z}$, with
$H_{\mathrm{dim}}(t,q)=\dim_{\mathrm{SU}(7)\times\mathrm{SU}(2)}H(t,q;\mathrm{SU}(7)\times\mathrm{SU}(2))$
and $H(t)=H_{\mathrm{dim}}(t,1)$. Likewise,
$\operatorname{PL}_{\mathrm{dim}}(t,q)=\dim_{\mathrm{SU}(7)\times\mathrm{SU}(2)}
\operatorname{PL}[H(t,q;\mathrm{SU}(7)\times\mathrm{SU}(2))]$ and
$\operatorname{PL}[H(t)]=\operatorname{PL}_{\mathrm{dim}}(t,1)$;
the latter agrees with the independently stored scalar result.


\section{Highest-weight generating function}
The source is Eq.~(12.3); the expression below is its $N=4$ specialization.
\begin{equation}
\mathrm{HWG}=\PE\!\left[(\mu_1\mu_6+\nu^2+1)t^2
+(\mu_2\mu_5+\nu q\mu_3+\nu q^{-1}\mu_4)t^4
+\mu_3\mu_4t^6-\nu^2\mu_3\mu_4t^8\right].
\end{equation}
\paragraph{Rational-product form.} This is the exact algebraic conversion of the specialized plethystic exponential.
\begin{equation}
\resizebox{0.98\linewidth}{!}{${\displaystyle
\mathrm{HWG}=\frac{1-\nu^2\mu_3\mu_4t^8}
{(1-t^2)(1-\mu_1\mu_6t^2)(1-\nu^2t^2)
 (1-\mu_2\mu_5t^4)(1-\nu q\mu_3t^4)
 (1-\nu q^{-1}\mu_4t^4)(1-\mu_3\mu_4t^6)}.}$}
\end{equation}
