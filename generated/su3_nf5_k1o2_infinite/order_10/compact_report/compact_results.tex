\maketitle
At infinite coupling the enhanced symmetry is $\mathrm{SU}(5)\times\mathrm{SU}(2)\times\mathrm{U}(1)_q$.
The expansion is truncated at $O(t^{11})$.  We define
$[a_1,\ldots,a_4;b]:=[a_1,\ldots,a_4]_{A_4}\otimes[b]_{A_1}$.
Dimension evaluation is the ring homomorphism
$\dim_{\mathrm{SU}(5)\times\mathrm{SU}(2)}:R(\mathrm{SU}(5)\times\mathrm{SU}(2))\to\mathbb{Z}$, with
$H_{\mathrm{dim}}(t,q)=\dim_{\mathrm{SU}(5)\times\mathrm{SU}(2)}H(t,q;\mathrm{SU}(5)\times\mathrm{SU}(2))$
and $H(t)=H_{\mathrm{dim}}(t,1)$.  Likewise,
$\operatorname{PL}_{\mathrm{dim}}(t,q)=\dim_{\mathrm{SU}(5)\times\mathrm{SU}(2)}
\operatorname{PL}[H(t,q;\mathrm{SU}(5)\times\mathrm{SU}(2))]$ and
$\operatorname{PL}[H(t)]=\operatorname{PL}_{\mathrm{dim}}(t,1)$;
the latter agrees with the independently stored scalar result.

\section{Highest-weight generating function}
\begin{equation}
\mathrm{HWG}=\PE\!\left[(\mu_1\mu_4+\nu^2+1)t^2
+\nu(q\mu_2+q^{-1}\mu_3)t^3
+\mu_2\mu_3t^4-\nu^2\mu_2\mu_3t^6\right].
\end{equation}
\begin{equation}
\resizebox{0.98\linewidth}{!}{${\displaystyle
\mathrm{HWG}=\frac{1-\nu^2\mu_2\mu_3t^6}
{(1-t^2)(1-\nu^2t^2)(1-\mu_1\mu_4t^2)
 (1-\nu q\mu_2t^3)(1-\nu q^{-1}\mu_3t^3)(1-\mu_2\mu_3t^4)}.}$}
\end{equation}
