At infinite coupling the enhanced symmetry is $\mathrm{SO}(10)\times\mathrm{U}(1)_q$.
The expansion is truncated at $O(t^{11})$, and
$[a_1,\ldots,a_{5}]$ denotes the $\mathrm{SO}(10)$ Dynkin label.
Dimension evaluation is the ring homomorphism
$\dim_{\mathrm{SO}(10)}:R(\mathrm{SO}(10))\to\mathbb{Z}$, with
$H_{\mathrm{dim}}(t,q)=\dim_{\mathrm{SO}(10)}H(t,q;\mathrm{SO}(10))$
and $H(t)=H_{\mathrm{dim}}(t,1)$.  Likewise,
$\operatorname{PL}_{\mathrm{dim}}(t,q)=\dim_{\mathrm{SO}(10)}
\operatorname{PL}[H(t,q;\mathrm{SO}(10))]$ and
$\operatorname{PL}[H(t)]=\operatorname{PL}_{\mathrm{dim}}(t,1)$;
the latter agrees with the independently stored scalar result.

\section{Highest-weight generating function}
\begin{equation}
\mathrm{HWG}=\PE\!\left[(\mu_2+1)t^2+(q\mu_4+q^{-1}\mu_5)t^3
+\mu_4\mu_5t^4-\mu_4\mu_5t^6\right].
\end{equation}
\begin{equation}
\resizebox{0.98\linewidth}{!}{${\displaystyle
\mathrm{HWG}=\frac{1-\mu_4\mu_5t^6}
{(1-t^2)(1-\mu_2t^2)(1-q\mu_4t^3)(1-q^{-1}\mu_5t^3)(1-\mu_4\mu_5t^4)}.}$}
\end{equation}
